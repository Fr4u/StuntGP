cmake_minimum_required(VERSION 3.25)
project(StuntGPCameraMounted LANGUAGES CXX)
enable_testing()
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(camera_mounted STATIC
 "${SGN_ROOT}/src/retail_camera_mounted.cpp" "${SGN_ROOT}/src/retail_camera.cpp"
 "${SGN_ROOT}/src/model_resources.cpp" "${SGN_ROOT}/src/source_pmd_model.cpp" "${SGN_ROOT}/src/source_allocation_bank.cpp" "${SGN_ROOT}/src/assets.cpp")
target_include_directories(camera_mounted PUBLIC "${SGN_ROOT}/include")
if(MSVC)
 target_compile_options(camera_mounted PUBLIC /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
else()
 target_compile_options(camera_mounted PUBLIC -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
add_executable(camera_mounted_oracle "${SGN_ROOT}/tests/camera_mounted_oracle.cpp")
target_link_libraries(camera_mounted_oracle PRIVATE camera_mounted)
add_test(NAME camera_mounted_source COMMAND camera_mounted_oracle
 "${SGN_ROOT}/reports/camera-work/series6/mounted-v1/mounted.dat" "${SGN_ROOT}/data/retail")
