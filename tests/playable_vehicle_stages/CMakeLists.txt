cmake_minimum_required(VERSION 3.25)
project(StuntGPPlayableVehicleStages LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SGN_BUILD_RUNTIME ON CACHE BOOL "" FORCE)
add_subdirectory("${SGN_ROOT}" root EXCLUDE_FROM_ALL)
add_executable(playable_vehicle_stages_contracts "${SGN_ROOT}/tests/playable_vehicle_stages_contracts.cpp" "${SGN_ROOT}/src/live_scene_geometry.cpp")
target_link_libraries(playable_vehicle_stages_contracts PRIVATE playable_driver playable_vehicle_stages race_overlay_textures)
add_test(NAME playable_vehicle_stages_contracts COMMAND playable_vehicle_stages_contracts "${SGN_ROOT}/data/retail")
add_executable(retail_camera_finish_session_tests "${SGN_ROOT}/tests/retail_camera_finish_session_tests.cpp")
target_link_libraries(retail_camera_finish_session_tests PRIVATE retail_camera_vehicle retail_vehicle_storage retail_trail_profiles retail_lighting retail_gameplay_configuration retail_game_text source_boot_config)
add_test(NAME retail_camera_finish_session_tests COMMAND retail_camera_finish_session_tests)
add_executable(playable_material_probe "${SGN_ROOT}/tests/playable_material_probe.cpp" "${SGN_ROOT}/src/live_scene_geometry.cpp")
target_link_libraries(playable_material_probe PRIVATE playable_driver playable_vehicle_stages race_overlay_textures)
add_executable(playable_frustum_probe "${SGN_ROOT}/tests/playable_frustum_probe.cpp" "${SGN_ROOT}/src/live_scene_geometry.cpp")
target_link_libraries(playable_frustum_probe PRIVATE playable_driver playable_vehicle_stages race_overlay_textures)
