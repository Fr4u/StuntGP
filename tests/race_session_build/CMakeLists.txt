cmake_minimum_required(VERSION 3.25)
project(StuntGPRaceSessionQualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SGN_ROOT "../.." ABSOLUTE)
set(SGN_BUILD_RUNTIME ON CACHE BOOL "" FORCE)
add_subdirectory("${SGN_ROOT}" root EXCLUDE_FROM_ALL)
add_library(race_session_candidate STATIC "${SGN_ROOT}/src/race_session.cpp")
# The candidate bypasses race_session, but its world still owns boot CFG.
target_link_libraries(race_session_candidate PUBLIC source_boot_config)
if(NOT TARGET track_part_animation)
  add_library(track_part_animation STATIC "${SGN_ROOT}/src/track_part_animation.cpp")
  target_link_libraries(track_part_animation PUBLIC retail_assets project_options)
endif()
if(NOT TARGET retail_world_view_prefix)
  add_library(retail_world_view_prefix STATIC "${SGN_ROOT}/src/retail_world_view_prefix.cpp")
  target_link_libraries(retail_world_view_prefix PUBLIC retail_camera retail_assets project_options)
endif()
if(NOT TARGET retail_viewport)
  add_library(retail_viewport STATIC "${SGN_ROOT}/src/retail_viewport.cpp")
  target_link_libraries(retail_viewport PUBLIC project_options)
endif()
target_link_libraries(race_session_candidate PUBLIC retail_race_initialization retail_game_text retail_gameplay_configuration
  retail_level_curves retail_level_lighting retail_color_tables retail_viewport retail_world_globals retail_world_view_prefix retail_model_matrices retail_trail_profiles track_part_animation retail_lighting scene_resources platform_audio vehicle_trails)
add_executable(race_session_tests "${SGN_ROOT}/tests/race_session_tests.cpp")
target_link_libraries(race_session_tests PRIVATE race_session_candidate)
add_executable(race_session_capture "${SGN_ROOT}/tests/race_session_capture.cpp")
target_link_libraries(race_session_capture PRIVATE race_session_candidate)
add_library(race_session_lifecycle_candidate STATIC "${SGN_ROOT}/src/race_session_lifecycle.cpp")
target_link_libraries(race_session_lifecycle_candidate PUBLIC race_session_candidate retail_time_trial_lifecycle retail_world)
add_executable(race_session_lifecycle_oracle "${SGN_ROOT}/tests/race_session_lifecycle_oracle.cpp")
target_link_libraries(race_session_lifecycle_oracle PRIVATE race_session_lifecycle_candidate)
add_executable(race_session_lifecycle_tests "${SGN_ROOT}/tests/race_session_lifecycle_tests.cpp")
target_link_libraries(race_session_lifecycle_tests PRIVATE race_session_lifecycle_candidate)
