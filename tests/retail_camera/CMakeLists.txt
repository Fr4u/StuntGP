cmake_minimum_required(VERSION 3.25)
project(StuntGPRetailCameraValidation LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(retail_camera STATIC "${SGN_ROOT}/src/retail_camera.cpp")
target_include_directories(retail_camera PUBLIC "${SGN_ROOT}/include")
if(MSVC)
  target_compile_options(retail_camera PUBLIC /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
else()
  target_compile_options(retail_camera PUBLIC -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
add_executable(retail_camera_oracle "${SGN_ROOT}/tests/retail_camera_oracle.cpp")
target_link_libraries(retail_camera_oracle PRIVATE retail_camera)
add_library(camera_contact_dependencies STATIC "${SGN_ROOT}/src/assets.cpp" "${SGN_ROOT}/src/track_contact.cpp" "${SGN_ROOT}/src/track_collision.cpp")
target_link_libraries(camera_contact_dependencies PUBLIC retail_camera)
add_library(retail_camera_collision STATIC "${SGN_ROOT}/src/retail_camera_collision.cpp")
target_link_libraries(retail_camera_collision PUBLIC retail_camera camera_contact_dependencies)
add_executable(retail_camera_collision_oracle "${SGN_ROOT}/tests/retail_camera_collision_oracle.cpp")
target_link_libraries(retail_camera_collision_oracle PRIVATE retail_camera_collision)
add_library(camera_vehicle_dependencies STATIC "${SGN_ROOT}/src/vehicle.cpp")
target_link_libraries(camera_vehicle_dependencies PUBLIC retail_camera camera_contact_dependencies)
add_library(retail_camera_vehicle STATIC "${SGN_ROOT}/src/retail_camera_vehicle.cpp")
target_link_libraries(retail_camera_vehicle PUBLIC retail_camera camera_vehicle_dependencies)
target_link_libraries(retail_camera_oracle PRIVATE retail_camera_vehicle)
add_executable(retail_camera_coupled_oracle "${SGN_ROOT}/tests/retail_camera_coupled_oracle.cpp")
target_link_libraries(retail_camera_coupled_oracle PRIVATE retail_camera_vehicle retail_camera_collision)
add_library(retail_camera_lifecycle STATIC "${SGN_ROOT}/src/retail_camera_lifecycle.cpp")
target_link_libraries(retail_camera_lifecycle PUBLIC retail_camera camera_contact_dependencies)
add_executable(retail_camera_lifecycle_oracle "${SGN_ROOT}/tests/retail_camera_lifecycle_oracle.cpp")
target_link_libraries(retail_camera_lifecycle_oracle PRIVATE retail_camera_lifecycle)
add_executable(retail_camera_scheduler_oracle "${SGN_ROOT}/tests/retail_camera_scheduler_oracle.cpp")
target_link_libraries(retail_camera_scheduler_oracle PRIVATE retail_camera_lifecycle retail_camera_vehicle)
add_library(retail_camera_transform STATIC "${SGN_ROOT}/src/retail_camera_transform.cpp")
target_link_libraries(retail_camera_transform PUBLIC retail_camera)
add_executable(retail_camera_transform_oracle "${SGN_ROOT}/tests/retail_camera_transform_oracle.cpp")
target_link_libraries(retail_camera_transform_oracle PRIVATE retail_camera_transform)
add_library(camera_spline_dependencies STATIC "${SGN_ROOT}/src/retail_spline.cpp")
target_link_libraries(camera_spline_dependencies PUBLIC camera_contact_dependencies)
add_library(camera_path STATIC "${SGN_ROOT}/src/camera_path.cpp")
target_link_libraries(camera_path PUBLIC retail_camera camera_spline_dependencies)
add_library(retail_camera_initialization STATIC "${SGN_ROOT}/src/retail_camera_initialization.cpp")
add_library(camera_matrix_dependencies STATIC "${SGN_ROOT}/src/retail_model_matrices.cpp" "${SGN_ROOT}/src/model_resources.cpp" "${SGN_ROOT}/src/source_pmd_model.cpp" "${SGN_ROOT}/src/source_allocation_bank.cpp")
target_link_libraries(camera_matrix_dependencies PUBLIC retail_camera camera_contact_dependencies)
target_link_libraries(retail_camera_initialization PUBLIC retail_camera_lifecycle camera_path camera_matrix_dependencies)
add_executable(camera_path_oracle "${SGN_ROOT}/tests/camera_path_oracle.cpp")
target_link_libraries(camera_path_oracle PRIVATE camera_path)
add_executable(retail_camera_initialization_oracle "${SGN_ROOT}/tests/retail_camera_initialization_oracle.cpp")
target_link_libraries(retail_camera_initialization_oracle PRIVATE retail_camera_initialization retail_camera_vehicle)
