cmake_minimum_required(VERSION 3.25)
project(StuntGPModelDrawQueueValidation LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DRAW_QUEUE_SOURCES retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets retail_world_view_prefix)
foreach(name IN LISTS DRAW_QUEUE_SOURCES)
 list(APPEND DRAW_QUEUE_FILES "${SGN_ROOT}/src/${name}.cpp")
endforeach()
add_library(draw_queue STATIC ${DRAW_QUEUE_FILES})
target_include_directories(draw_queue PUBLIC "${SGN_ROOT}/include")
if(MSVC)
 target_compile_options(draw_queue PUBLIC /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
else()
 target_compile_options(draw_queue PUBLIC -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
add_executable(retail_model_draw_queue_oracle "${SGN_ROOT}/tests/retail_model_draw_queue_oracle.cpp")
target_link_libraries(retail_model_draw_queue_oracle PRIVATE draw_queue)
enable_testing()
add_test(NAME retail_model_draw_queue_source COMMAND retail_model_draw_queue_oracle
 "${SGN_ROOT}/reports/camera-work/series2/model-draw-queue/qualified/queue.dat" "${SGN_ROOT}/data/retail")
