cmake_minimum_required(VERSION 3.25)
project(StuntGPModelMatrixValidation LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(model_matrices STATIC "${SGN_ROOT}/src/retail_model_matrices.cpp" "${SGN_ROOT}/src/retail_camera.cpp"
 "${SGN_ROOT}/src/model_resources.cpp" "${SGN_ROOT}/src/source_pmd_model.cpp" "${SGN_ROOT}/src/source_allocation_bank.cpp" "${SGN_ROOT}/src/assets.cpp")
target_include_directories(model_matrices PUBLIC "${SGN_ROOT}/include")
if(MSVC)
 target_compile_options(model_matrices PUBLIC /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
else()
 target_compile_options(model_matrices PUBLIC -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
add_executable(retail_model_matrices_oracle "${SGN_ROOT}/tests/retail_model_matrices_oracle.cpp")
target_link_libraries(retail_model_matrices_oracle PRIVATE model_matrices)
add_executable(retail_model_matrices_contracts "${SGN_ROOT}/tests/retail_model_matrices_contracts.cpp")
target_link_libraries(retail_model_matrices_contracts PRIVATE model_matrices)
enable_testing()
add_test(NAME retail_model_matrices_source COMMAND retail_model_matrices_oracle
 "${SGN_ROOT}/reports/camera-work/series2/model-matrices/matrices.dat" "${SGN_ROOT}/data/retail")
add_test(NAME retail_model_matrices_contracts COMMAND retail_model_matrices_contracts)
