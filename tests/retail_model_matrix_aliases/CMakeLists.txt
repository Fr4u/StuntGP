cmake_minimum_required(VERSION 3.25)
project(StuntGPModelMatrixAliasValidation LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(retail_model_matrix_aliases "${SGN_ROOT}/tests/retail_model_matrix_aliases.cpp"
 "${SGN_ROOT}/src/retail_model_matrices.cpp" "${SGN_ROOT}/src/retail_camera.cpp" "${SGN_ROOT}/src/model_resources.cpp" "${SGN_ROOT}/src/source_pmd_model.cpp" "${SGN_ROOT}/src/source_allocation_bank.cpp" "${SGN_ROOT}/src/assets.cpp")
target_include_directories(retail_model_matrix_aliases PRIVATE "${SGN_ROOT}/include")
if(MSVC)
 target_compile_options(retail_model_matrix_aliases PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
else()
 target_compile_options(retail_model_matrix_aliases PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
enable_testing()
add_test(NAME retail_model_matrix_aliases COMMAND retail_model_matrix_aliases "${SGN_ROOT}/reports/camera-work/series2/model-matrix-aliases/aliases.dat")
