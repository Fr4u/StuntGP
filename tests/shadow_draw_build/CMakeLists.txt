cmake_minimum_required(VERSION 3.25)
project(StuntGPShadowDraw LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SHADOW_DRAW_SOURCES vehicle_shadow_draw retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets retail_world_view_prefix)
foreach(name IN LISTS SHADOW_DRAW_SOURCES)
  list(APPEND SHADOW_DRAW_FILES "${SGN_ROOT}/src/${name}.cpp")
endforeach()
add_executable(shadow_draw_oracle "${SGN_ROOT}/tests/shadow_draw_oracle.cpp" ${SHADOW_DRAW_FILES})
target_include_directories(shadow_draw_oracle PRIVATE "${SGN_ROOT}/include")
target_compile_options(shadow_draw_oracle PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
