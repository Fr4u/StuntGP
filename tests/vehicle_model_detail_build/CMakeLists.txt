cmake_minimum_required(VERSION 3.25)
project(StuntGPModelDetail LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(vehicle_model_detail_oracle "${SGN_ROOT}/tests/vehicle_model_detail_oracle.cpp" "${SGN_ROOT}/src/vehicle_model_detail.cpp" "${SGN_ROOT}/src/model_resources.cpp" "${SGN_ROOT}/src/source_pmd_model.cpp" "${SGN_ROOT}/src/source_allocation_bank.cpp" "${SGN_ROOT}/src/vehicle_presentation.cpp" "${SGN_ROOT}/src/assets.cpp")
target_include_directories(vehicle_model_detail_oracle PRIVATE "${SGN_ROOT}/include")
if(MSVC)
 target_compile_options(vehicle_model_detail_oracle PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
 target_compile_options(vehicle_model_detail_oracle PRIVATE /Gy)
 target_link_options(vehicle_model_detail_oracle PRIVATE /OPT:REF)
endif()
