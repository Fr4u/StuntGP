cmake_minimum_required(VERSION 3.25)
project(StuntGPVehicleModelPrepareFull LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PREPARE_SOURCES vehicle_model_prepare vehicle_model_pose vehicle_model_basis vehicle_segments retail_camera model_resources source_allocation_bank source_pmd_model assets retail_world_view_prefix)
foreach(name IN LISTS PREPARE_SOURCES)
  list(APPEND PREPARE_FILES "${SGN_ROOT}/src/${name}.cpp")
endforeach()
add_executable(vehicle_model_prepare_full "${SGN_ROOT}/tests/vehicle_model_prepare_full.cpp" ${PREPARE_FILES})
target_include_directories(vehicle_model_prepare_full PRIVATE "${SGN_ROOT}/include")
target_compile_options(vehicle_model_prepare_full PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
enable_testing()
if(NOT PREPARE_CORPUS)
  message(FATAL_ERROR "PREPARE_CORPUS must name the generated source corpus")
endif()
add_test(NAME vehicle_model_prepare_full COMMAND vehicle_model_prepare_full "${PREPARE_CORPUS}" "${SGN_ROOT}/data/retail")
set_tests_properties(vehicle_model_prepare_full PROPERTIES TIMEOUT 90)
