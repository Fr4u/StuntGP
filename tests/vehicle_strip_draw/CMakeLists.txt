cmake_minimum_required(VERSION 3.25)
project(StuntGPStripDraw LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(vehicle_strip_draw "${SGN_ROOT}/tests/vehicle_strip_draw.cpp")
set(STRIP_SOURCES retail_vehicle_strip_draw retail_vehicle_strip retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets)
foreach(name IN LISTS STRIP_SOURCES)
 target_sources(vehicle_strip_draw PRIVATE "${SGN_ROOT}/src/${name}.cpp")
endforeach()
target_include_directories(vehicle_strip_draw PRIVATE "${SGN_ROOT}/include")
target_compile_features(vehicle_strip_draw PRIVATE cxx_std_23)
target_compile_options(vehicle_strip_draw PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
enable_testing()
add_test(NAME strip_draw_binding_contracts COMMAND vehicle_strip_draw --contracts)
