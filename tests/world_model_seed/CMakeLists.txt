cmake_minimum_required(VERSION 3.25)
project(StuntGPWorldModelSeedValidation LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SEED_SOURCES world_model_draw retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets retail_world_view_prefix)
foreach(name IN LISTS SEED_SOURCES)
 list(APPEND SEED_FILES "${SGN_ROOT}/src/${name}.cpp")
endforeach()
add_library(world_seed STATIC ${SEED_FILES})
target_include_directories(world_seed PUBLIC "${SGN_ROOT}/include")
if(MSVC)
 target_compile_options(world_seed PUBLIC /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
else()
 target_compile_options(world_seed PUBLIC -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
add_executable(world_model_seed_oracle "${SGN_ROOT}/tests/world_model_seed_oracle.cpp")
target_link_libraries(world_model_seed_oracle PRIVATE world_seed)
