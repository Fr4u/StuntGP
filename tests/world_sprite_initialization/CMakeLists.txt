cmake_minimum_required(VERSION 3.25)
project(StuntGPSpriteInitialization LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(world_sprite_initialization "${SGN_ROOT}/tests/world_sprite_initialization.cpp")
set(SPRITE_SOURCES retail_world_sprite_model world_model_draw retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets)
foreach(name IN LISTS SPRITE_SOURCES)
 target_sources(world_sprite_initialization PRIVATE "${SGN_ROOT}/src/${name}.cpp")
endforeach()
target_include_directories(world_sprite_initialization PRIVATE "${SGN_ROOT}/include")
target_compile_features(world_sprite_initialization PRIVATE cxx_std_23)
target_compile_options(world_sprite_initialization PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
enable_testing()
add_test(NAME sprite_initialization_contracts COMMAND world_sprite_initialization --contracts)
