cmake_minimum_required(VERSION 3.25)
project(GeneratedOwnerBindings LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT MSVC)
  message(FATAL_ERROR "MSVC required for current qualification")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  add_executable(generated_owner_original ../generated_owner_original.cpp)
  target_link_libraries(generated_owner_original PRIVATE bcrypt)
  target_link_options(generated_owner_original PRIVATE /BASE:0x50000000 /DYNAMICBASE:NO /FIXED /INCREMENTAL:NO)
  set(TARGET_NAME generated_owner_original)
else()
  add_executable(generated_owner_native ../generated_owner_native.cpp)
  set(OWNER_SOURCES retail_world_sprite_model retail_vehicle_strip retail_world_sprite_producer world_model_draw retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets retail_camera_transform retail_world_polygon_producer retail_world_local_camera)
  foreach(name IN LISTS OWNER_SOURCES)
    target_sources(generated_owner_native PRIVATE "${SGN_ROOT}/src/${name}.cpp")
  endforeach()
  target_include_directories(generated_owner_native PRIVATE "${SGN_ROOT}/include")
  set(TARGET_NAME generated_owner_native)
  add_executable(generated_owner_legacy_normal ../generated_owner_native.cpp ../generated_owner_legacy_normal.cpp)
  foreach(name IN LISTS OWNER_SOURCES)
    if(NOT name STREQUAL "retail_world_sprite_producer")
      target_sources(generated_owner_legacy_normal PRIVATE "${SGN_ROOT}/src/${name}.cpp")
    endif()
  endforeach()
  target_include_directories(generated_owner_legacy_normal PRIVATE "${SGN_ROOT}/include")
  target_compile_features(generated_owner_legacy_normal PRIVATE cxx_std_23)
  target_compile_options(generated_owner_legacy_normal PRIVATE /W4 /WX /fp:strict /permissive- /utf-8 /EHsc)
  enable_testing()
  add_test(NAME generated_owner_bindings COMMAND generated_owner_native --contracts)
endif()
target_compile_features(${TARGET_NAME} PRIVATE cxx_std_23)
target_compile_options(${TARGET_NAME} PRIVATE /W4 /WX /fp:strict /permissive- /utf-8 /EHsc)
