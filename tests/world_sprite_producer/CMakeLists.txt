cmake_minimum_required(VERSION 3.25)
project(StuntGPSpriteProducer LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(world_sprite_producer "${SGN_ROOT}/tests/world_sprite_producer.cpp")
set(SPRITE_SOURCES retail_world_sprite_producer world_model_draw retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets retail_camera_transform retail_world_polygon_producer retail_world_local_camera)
foreach(name IN LISTS SPRITE_SOURCES)
 target_sources(world_sprite_producer PRIVATE "${SGN_ROOT}/src/${name}.cpp")
endforeach()
target_include_directories(world_sprite_producer PRIVATE "${SGN_ROOT}/include")
target_compile_features(world_sprite_producer PRIVATE cxx_std_23)
target_compile_options(world_sprite_producer PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
enable_testing()
add_test(NAME sprite_producer_binding_contracts COMMAND world_sprite_producer --contracts)
