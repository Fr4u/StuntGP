cmake_minimum_required(VERSION 3.25)
project(StuntGPLiveTemplateViews LANGUAGES CXX)
get_filename_component(SGN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(template_views STATIC)
foreach(name vehicle_draw_pose vehicle_presentation vehicle_shadow_draw retail_model_draw_queue retail_model_matrices retail_camera model_resources source_allocation_bank source_pmd_model assets retail_world_view_prefix)
 target_sources(template_views PRIVATE "${SGN_ROOT}/src/${name}.cpp")
endforeach()
target_include_directories(template_views PUBLIC "${SGN_ROOT}/include")
target_compile_options(template_views PUBLIC /W4 /WX /permissive- /utf-8 /EHsc /fp:strict)
foreach(name world_template_draw_pose world_template_cold_presentation world_template_shadow_draw)
 add_executable(${name} "${SGN_ROOT}/tests/${name}.cpp")
 target_link_libraries(${name} PRIVATE template_views)
endforeach()
enable_testing()
add_test(NAME cold_presentation_contracts COMMAND world_template_cold_presentation --contracts)
